<script>
function updateStatus(){
  fetch('/status').then(r=>r.json()).then(js=>{
    let st = document.getElementById('status');
    if(js.connected){
      st.innerHTML = 'Connected to: <b>'+js.ssid+'</b> — IP: '+js.ip;
    } else {
      st.innerHTML = '<b>Not connected</b>';
    }
  }).catch(e=>{
    document.getElementById('status').innerText = 'Status not available';
  });
}
function scanAps(){
  fetch('/aplist').then(r=>r.json()).then(list=>{
    let cont = document.getElementById('aplist');
    cont.innerHTML = '';
    if(list.length==0){ cont.innerText = 'No visible networks found.'; return; }
    list.sort((a,b)=>b.rssi - a.rssi);
    list.forEach(item=>{
      let div = document.createElement('div');
      div.className = 'ap';
      div.innerHTML = '<div><b>'+item.ssid+'</b><div class="small">RSSI: '+item.rssi+'</div></div>';
      let btn = document.createElement('button');
      btn.textContent = 'Select';
      btn.onclick = ()=>{ document.getElementById('ssid').value = item.ssid; };
      div.appendChild(btn);
      cont.appendChild(div);
    });
  }).catch(e=>{
    document.getElementById('aplist').innerText = 'Failed to fetch AP list.';
  });
}
document.getElementById('saveBtn').onclick = function(){
  let ssid = document.getElementById('ssid').value;
  let pass = document.getElementById('password').value;
  if(!ssid){ alert('Please enter or select an SSID'); return; }
  fetch('/connect', {method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({ssid:ssid,password:pass})})
    .then(r=>r.json()).then(js=>{
      alert(js.message || (js.ok? 'Attempting connection' : 'Failed'));
      updateStatus();
    }).catch(e=>{ alert('Error attempting to connect'); });
};
window.addEventListener('load', ()=>{
  updateStatus();
  scanAps();
  setInterval(scanAps, 10000);
  setInterval(updateStatus, 5000);
});
</script>
